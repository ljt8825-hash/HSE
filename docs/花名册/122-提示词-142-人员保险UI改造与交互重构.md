```javascript
/**
 * GET /personnel/departments
 * 返回人员保险表中实际存在的部门列表（去重、非空）
 */
router.get('/personnel/departments', async (req, res) => {
  try {
    let departments = [];
    if (isConnected()) {
      const r = await pool.query(`
        SELECT DISTINCT department 
        FROM personnel_insurance 
        WHERE is_deleted = false 
          AND department IS NOT NULL 
          AND department <> ''
        ORDER BY department ASC
      `);
      departments = r.rows.map(row => row.department);
    } else {
      departments = [...new Set(
        memoryStores.personnel
          .filter(p => !p.is_deleted && p.department)
          .map(p => p.department)
      )].sort();
    }
    res.json(success({ departments }));
  } catch (err) {
    res.json(error(-1, err.message));
  }
});
```

### 4.7 R6：身份证号格式校验（P2）

**问题**：人员保险新增/编辑时，身份证号无格式校验，可录入任意字符串。

**改造内容**：

1. **前端校验函数**：
```typescript
function isValidIdCard(idCard: string): boolean {
  if (!idCard) return false;
  // 18位身份证基础校验
  const reg = /^[1-9]\d{5}(18|19|20)\d{2}((0[1-9])|(1[0-2]))(([0-2][1-9])|10|20|30|31)\d{3}[0-9Xx]$/;
  if (!reg.test(idCard)) return false;
  // 校验位校验（可选，如项目已有工具函数优先复用）
  return true;
}
```

2. **表单提交校验**（`handleSavePerson`开头）：
```typescript
async function handleSavePerson() {
  if (!personForm.name || !personForm.idCard) {
    message.warning('请填写必填项');

  }
  if (!isValidIdCard(personForm.idCard)) {
    message.warning('身份证号格式不正确');
    return;
  }
  // ...原有逻辑
}
```

> 如项目中已有通用身份证校验工具（如 `@/utils/validator`），优先复用。
