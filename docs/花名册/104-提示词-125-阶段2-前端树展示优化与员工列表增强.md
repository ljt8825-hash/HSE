```typescript
function orgTypeText(orgType?: string) {
  const map: Record<string, string> = {
    company: '公司',
    department: '职能部门',
    center: '作业中心',
    town_center: '乡镇',
    team: '班组',
  };
  return orgType ? map[orgType] || orgType : '';
}

function orgTypeColor(orgType?: string) {
  const map: Record<string, string> = {
    company: 'red',
    department: 'blue',
    center: 'orange',
    town_center: 'purple',
    team: 'green',
  };
  return orgType ? map[orgType] || 'default' : 'default';
}
```

### 4.3 修改项 C：hse-employee.js — `/departments/:id/employees` 增强

**文件**：`server\routes\hse-employee.js`

#### C-1：处理 ou_X 合成 ID

当前代码 `const deptId = parseInt(req.params.id)` 对 `"ou_5"` 返回 `NaN`，导致后续查询失败。

**修复逻辑**（在 `parseInt` 之前插入）：

```javascript
const rawId = req.params.id;
let deptId;

// 处理 ou_X 合成 ID（来自 hse_org_unit 节点）
if (typeof rawId === 'string' && rawId.startsWith('ou_')) {
  const orgUnitId = parseInt(rawId.slice(3), 10);
  if (isNaN(orgUnitId)) {
    return res.json(error(-2, '无效的组织ID'));
  }
  // 直接按 org_unit_id 查询员工
  const { page = 1, pageSize = 20 } = req.query;
  const pageNum = Math.max(1, parseInt(page) || 1);
  const limitNum = Math.min(Math.max(1, parseInt(pageSize) || 10), 200);
  const offset = (pageNum - 1) * limitNum;

  const countResult = await pool.query(
    `SELECT COUNT(*)::int AS total FROM employee e
     JOIN employee_assignment ea ON ea.employee_id = e.id AND ea.is_current = true AND ea.is_deleted = false AND ea.is_primary = true
     WHERE e.is_deleted = false AND ea.org_unit_id = $1`,
    [orgUnitId]
  );
  const total = countResult.rows[0].total;

  const result = await pool.query(
    `SELECT e.id, e.employee_code, e.id_card, e.name, e.gender, e.birth_date, e.phone,
            CASE WHEN e.birth_date IS NOT NULL THEN EXTRACT(YEAR FROM age(e.birth_date))::INT ELSE NULL END AS age,
            ou.org_path, ea.position_id, ea.start_date,
            p.name AS position_name
     FROM employee e
     JOIN employee_assignment ea ON ea.employee_id = e.id AND ea.is_current = true AND ea.is_deleted = false AND ea.is_primary = true
     JOIN hse_org_unit ou ON ea.org_unit_id = ou.id AND ou.is_deleted = false
     LEFT JOIN position p ON ea.position_id = p.id AND p.is_deleted = false
     WHERE e.is_deleted = false AND ea.org_unit_id = $1
     ORDER BY e.created_at DESC
     LIMIT $2 OFFSET $3`,
    [orgUnitId, limitNum, offset]
  );

  return buildListResponse(res, result.rows, total, pageNum, limitNum, '查询成功', { camelCase: false });
}

deptId = parseInt(rawId);
```

#### C-2：班组节点名称前缀匹配增强

当前班组节点（如 `长官镇一班`）在 `department` 表中 `org_unit_id` 为 NULL，回退到名称前缀匹配时会用 `长官镇一班` 去 `hse_org_unit` 匹配 `org_name`，无法匹配到 `长官镇`。

**修复逻辑**（在现有 `org_unit_id` 为空的分支中增加乡镇前缀提取）：

在现有代码的降级分支（`orgUnitIds` 为空时）中，增加班组名称前缀提取逻辑：

```javascript
// 降级：未映射的部门使用字符串前缀匹配
const deptName = deptRow.rows[0].name;

// 班组名称提取乡镇前缀（如 "长官镇一班" → "长官镇"）
let matchName = deptName;
const teamPattern = /^(.+?)(一|二|三|四|五|六|七|八|九|十)班$/;
const areaPattern = /^(.+?)片区$/;
const fleetMatch = deptName === '车队';

if (teamPattern.test(deptName)) {
  matchName = deptName.match(teamPattern)[1];
} else if (areaPattern.test(deptName)) {
  matchName = deptName.match(areaPattern)[1];
} else if (fleetMatch) {
  matchName = deptName; // 车队保持原名
}

const ouRows = await pool.query(
  `SELECT id, (org_name = $1) AS exact_hit FROM hse_org_unit
    WHERE is_deleted = false AND (org_name = $1 OR org_name LIKE $1 || '%')`,
  [matchName]
);
```
