### 2. `server/routes/hse-employee-import-v2.js` → `POST /clear-all`

**任务3 — 中文外键错误友好提示（兼容中英文 PG 错误消息）**
```javascript
} catch (err) {
  await client.query('ROLLBACK');
  const msg = String(err.message || err);
  if (msg.includes('foreign key') || msg.includes('violates foreign key constraint') || msg.includes('外键约束') || msg.includes('违反')) {
    const m = msg.match(/(?:违反了在|violates) "([^"]+)" 上的外键约束|on table "([^"]+)"/i);
    const tbl = (m && (m[1] || m[2])) || '';
    const detail = tbl ? `（被表 ${tbl} 引用）` : '';
    throw new Error(`清空失败：员工数据被其他业务表引用${detail}，请先清理相关业务数据后再执行清空`);
  }
  throw err;
} finally {
  client.release();
}
```
> 关键修正：本库 PostgreSQL locale 为中文，FK 报错为
> `在 "employee" 上的更新或删除操作违反了在 "risk_level_records" 上的外键约束`，
> 不含英文 `foreign key` 串。首版仅匹配英文导致友好提示未触发；本版新增
> `外键约束`/`违反` 关键词并正确提取**子表（引用表）**名。
