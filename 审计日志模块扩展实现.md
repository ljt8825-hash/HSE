# 审计日志模块扩展实现

## 1. 架构概述

审计日志模块是安全管理系统最核心的扩展功能之一，用于记录和追踪系统内的所有关键操作。本模块完全基于已有的 Feature-Sliced Design 架构扩展，遵循以下设计原则：

- **TanStack Query 管理服务端状态**：所有数据获取/变更通过 Vue Query 封装，Pinia 仅保留客户端 UI 状态
- **权限驱动**：菜单项和按钮级操作均通过 `permissionStore` + `v-permission` 指令控制
- **类型安全**：所有 API 参数和响应均使用 TypeScript 接口约束
- **暗色模式兼容**：所有样式使用 Element Plus CSS 变量，自动适配 `.dark` 主题

## 2. 目录结构

```
src/features/audit/
  views/AuditLogView.vue          # 审计日志主页面（筛选 + 表格组装）
  components/AuditLogTable.vue    # 日志表格 + 统计卡片 + 详情弹窗
  composables/useAuditLog.ts      # TanStack Query 封装（查询、导出、实时刷新）
  types.ts                        # 模块私有类型定义

src/api/modules/audit.ts          # 审计日志 API 接口
src/api/keys.ts                   # 新增 auditKeys Query Key Factory
src/router/routes.ts              # 新增 /system/audit 路由
```

## 3. 功能清单

| 功能 | 实现位置 | 技术要点 |
|------|---------|---------|
| 分页查询日志 | `useAuditLogs` + `AuditLogTable` | `placeholderData` 保持翻页流畅 |
| 多维度筛选 | `AuditLogView.vue` | 时间范围、操作人、操作类型、状态、关键词 |
| 排序 | `el-table @sort-change` | 后端排序，支持时间和耗时字段 |
| 统计概览 | `useAuditLogStats` | 总数、今日、失败次数、平均耗时 |
| 导出 Excel | `useExportAuditLogs` | Blob 下载，文件名带时间戳 |
| 实时刷新 | `useAuditRealtime` | 轮询降级，预留 WebSocket 接口 |
| 日志详情 | `el-dialog` + `el-descriptions` | JSON 参数/响应格式化展示 |
| 权限控制 | `v-permission="'audit:export'"` | 按钮级 RBAC |

### `src/features/audit/composables/useAuditLog.ts`

```ts
/**
 * features/audit/composables/useAuditLog.ts
 * 审计日志数据层 —— TanStack Query 封装
 */

import { computed, ref, type Ref } from 'vue'
import { useQuery, useMutation, useQueryClient } from '@tanstack/vue-query'
import {
  fetchAuditLogs,
  fetchAuditLogStats,
  exportAuditLogs,
  fetchAuditOperators,
  fetchAuditActionTypes,
} from '@/api/modules/audit'
import { auditKeys } from '@/api/keys'
import type {
  AuditLog,
  AuditLogQueryParams,
  AuditLogStats,
  ExportAuditLogParams,
} from '../types'
import type { PageResult } from '@/types/api'

export function useAuditLogs(params: Ref<AuditLogQueryParams>) {
  return useQuery<PageResult<AuditLog>>({
    queryKey: computed(() => auditKeys.list(params.value)),
    queryFn: () => fetchAuditLogList(params.value),
    staleTime: 30_000,
    placeholderData: (previousData) => previousData,
  })
}

async function fetchAuditLogList(p: AuditLogQueryParams): Promise<PageResult<AuditLog>> {
  const res = await fetchAuditLogs(p)
  return res.data
}

export function useAuditLogStats(
  dateRange: Ref<{ startDate?: string; endDate?: string }>
) {
  return useQuery<AuditLogStats>({
    queryKey: computed(() => auditKeys.stats(dateRange.value)),
    queryFn: () => fetchAuditStats(dateRange.value),
    staleTime: 60_000,
  })
}

async function fetchAuditStats(p: {
  startDate?: string
  endDate?: string
}): Promise<AuditLogStats> {
  const res = await fetchAuditLogStats(p)
  return res.data
}

export function useAuditOperators() {
  return useQuery({
    queryKey: auditKeys.operators(),
    queryFn: async () => {
      const res = await fetchAuditOperators()
      return res.data
    },
    staleTime: 300_000,
  })
}

export function useAuditActionTypes() {
  return useQuery({
    queryKey: auditKeys.actionTypes(),
    queryFn: async () => {
      const res = await fetchAuditActionTypes()
      return res.data
    },
    staleTime: 300_000,
  })
}

export function useExportAuditLogs() {
  return useMutation({
    mutationFn: (params: ExportAuditLogParams) => exportAuditLogs(params),
    onSuccess: (blob, variables) => {
      const url = window.URL.createObjectURL(blob)
      const link = document.createElement('a')
      link.href = url
      const ext = variables.format === 'xlsx' ? 'xlsx' : 'csv'
      const timestamp = new Date().toISOString().slice(0, 10)
      link.download = `审计日志_${timestamp}.${ext}`
      document.body.appendChild(link)
      link.click()
      document.body.removeChild(link)
      window.URL.revokeObjectURL(url)
    },
  })
}

export function useAuditRealtime(
  enabled: Ref<boolean> = ref(false)
) {
  const queryClient = useQueryClient()

  // TODO: WebSocket 连接成功后，接收新日志推送并 invalidate query
  // const ws = useWebSocket('/ws/audit', {
  //   onMessage: (msg) => {
  //     if (msg.type === 'NEW_AUDIT_LOG') {
  //       queryClient.invalidateQueries({ queryKey: auditKeys.all })
  //     }
  //   },
  // })

  // 当前使用轮询作为降级方案
  const { isFetching } = useQuery({
    queryKey: auditKeys.realtime(),
    queryFn: async () => {
      return Date.now()
    },
    refetchInterval: enabled.value ? 30_000 : false,
    enabled: computed(() => enabled.value),
  })

  return {
    isRealtimeEnabled: enabled,
    toggleRealtime: () => {
      enabled.value = !enabled.value
    },
    isRefreshing: isFetching,
    refresh: () => {
      queryClient.invalidateQueries({ queryKey: auditKeys.all })
    },
  }
}
```

```vue
<script setup lang="ts">
import { ref, computed } from 'vue'
import type { AuditLog, AuditLogStats, AuditLogQueryParams } from '../types'

interface Props {
  logs: AuditLog[]
  total: number
  loading?: boolean
  stats?: AuditLogStats
  page: number
  pageSize: number
}

const props = withDefaults(defineProps<Props>(), {
  loading: false,
})

const emit = defineEmits<{
  (e: 'update:page', page: number): void
  (e: 'update:pageSize', size: number): void
  (e: 'sortChange', sort: { field: string; order: 'asc' | 'desc' }): void
}>()

const currentPage = computed({
  get: () => props.page,
  set: (val) => emit('update:page', val),
})

const pageSize = computed({
  get: () => props.pageSize,
  set: (val) => emit('update:pageSize', val),
})

const detailVisible = ref(false)
const selectedLog = ref<AuditLog | null>(null)

function handlePageChange() {}

function handleSortChange({ prop, order }: { prop: string; order: string | null }) {
  if (!prop || !order) return
  emit('sortChange', {
    field: prop,
    order: order === 'ascending' ? 'asc' : 'desc',
  })
}

function handleDetail(row: AuditLog) {
  selectedLog.value = row
  detailVisible.value = true
}

function formatNumber(n?: number): string {
  if (n === undefined || n === null) return '-'
  return n.toLocaleString('zh-CN')
}

function formatDateTime(iso: string): string {
  const d = new Date(iso)
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')} ${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}:${String(d.getSeconds()).padStart(2, '0')}`
}

function actionTypeTag(type: string): string {
  const map: Record<string, string> = {
    CREATE: 'success', UPDATE: 'primary', DELETE: 'danger',
    LOGIN: 'info', LOGOUT: 'info', EXPORT: 'warning',
    IMPORT: 'warning', APPROVE: 'success', REJECT: 'danger',
    VIEW: '', OTHER: 'info',
  }
  return map[type] || 'info'
}

function actionTypeText(type: string): string {
  const map: Record<string, string> = {
    CREATE: '创建', UPDATE: '更新', DELETE: '删除',
    LOGIN: '登录', LOGOUT: '登出', EXPORT: '导出',
    IMPORT: '导入', APPROVE: '审批', REJECT: '驳回',
    VIEW: '查看', OTHER: '其他',
  }
  return map[type] || type
}
```

```vue
import { ref, computed } from 'vue'

  useAuditLogStats,
  useAuditOperators,
  useAuditActionTypes,
  useExportAuditLogs,
  useAuditRealtime,

import type { AuditLogQueryParams } from '../types'

interface FilterForm {
  dateRange: [string, string] | null
  operatorId: number | null
  actionType: string | null
  status: 'success' | 'failure' | null
  keyword: string
}

const filterForm = ref<FilterForm>({
  dateRange: null, operatorId: null, actionType: null, status: null, keyword: '',
})

const queryParams = ref<AuditLogQueryParams>({
  page: 1, pageSize: 20,
  sortField: 'createdAt', sortOrder: 'desc',
})

const logsQuery = useAuditLogs(queryParams)
const statsQuery = useAuditLogStats(
  computed(() => ({
    startDate: queryParams.value.startDate,
    endDate: queryParams.value.endDate,
  }))
)
const operatorsQuery = useAuditOperators()
const actionTypesQuery = useAuditActionTypes()
const exportMutation = useExportAuditLogs()
const { isRealtimeEnabled: realtimeEnabled, toggleRealtime } = useAuditRealtime(ref(false))

const logs = computed(() => logsQuery.data.value?.list ?? [])
const total = computed(() => logsQuery.data.value?.total ?? 0)
const logsLoading = computed(() => logsQuery.isLoading.value)
const stats = computed(() => statsQuery.data.value)
const operators = computed(() => operatorsQuery.data.value ?? [])
const actionTypes = computed(() => actionTypesQuery.data.value ?? [])
const exporting = computed(() => exportMutation.isPending.value)

function handleSearch() {
  queryParams.value.page = 1
  queryParams.value.startDate = filterForm.value.dateRange?.[0] || undefined
  queryParams.value.endDate = filterForm.value.dateRange?.[1] || undefined
  queryParams.value.operatorId = filterForm.value.operatorId || undefined
  queryParams.value.actionType = (filterForm.value.actionType as any) || undefined
  queryParams.value.status = filterForm.value.status || undefined
  queryParams.value.keyword = filterForm.value.keyword || undefined
}

function handleReset() {
  filterForm.value = { dateRange: null, operatorId: null, actionType: null, status: null, keyword: '' }
  queryParams.value = {
    page: 1, pageSize: 20,
    startDate: undefined, endDate: undefined,
    operatorId: undefined, actionType: undefined,
    status: undefined, keyword: undefined,
    sortField: 'createdAt', sortOrder: 'desc',
  }
}

function handleSortChange(sort: { field: string; order: 'asc' | 'desc' }) {
  queryParams.value.sortField = sort.field
  queryParams.value.sortOrder = sort.order
}

function handleExport() {
  exportMutation.mutate({
    format: 'xlsx',
    startDate: queryParams.value.startDate,
    endDate: queryParams.value.endDate,
    operatorId: queryParams.value.operatorId,
    actionType: queryParams.value.actionType,
    status: queryParams.value.status,
    keyword: queryParams.value.keyword,
  })
}
```

## 8. 数据流说明

```
用户操作筛选条件
    ↓
AuditLogView.filterForm（本地 UI 状态）
    ↓
handleSearch() → 同步到 queryParams（Vue Query 观察的 Ref）
    ↓
useAuditLogs(queryParams) → TanStack Query 自动重新请求
    ↓
auditKeys.list(params) 匹配 / 失效 / 发起请求
    ↓
fetchAuditLogs() → Axios GET /audit/logs
    ↓
api/client.ts 拦截器附加 Token + 错误处理
    ↓
返回 PageResult<AuditLog> → 渲染 AuditLogTable
```
