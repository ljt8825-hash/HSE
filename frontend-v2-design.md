# 前端页面设计

## 一、设计原则

### 1.1 Vben Admin 规范

- 遵循 Vben Admin v5.7.0 组件规范
- 使用 Ant Design Vue 组件库
- 统一布局和配色方案
- 响应式设计

### 1.2 设计风格

- 专业、简洁的企业级后台风格
- 使用蓝色系作为主色调
- 卡片式布局
- 清晰的数据可视化

### 1.3 交互原则

- 操作反馈及时
- 批量操作支持
- 拖拽排序
- 搜索过滤

// src/api/hse/document.ts
import { defHttp } from '/@/utils/http/axios';

export const documentAPI = {
  upload: (data: FormData) => 
    defHttp.post('/api/hse/documents/upload', { data, isFormData: true }),
  
  list: (params: DocumentListParams) => 
    defHttp.get('/api/hse/documents', { params }),
  
  detail: (id: number) => 
    defHttp.get(`/api/hse/documents/${id}`),
  
  delete: (id: number) => 
    defHttp.delete(`/api/hse/documents/${id}`),
  
  download: (id: number) => 
    defHttp.get(`/api/hse/documents/${id}/download`, { responseType: 'blob' })
};
