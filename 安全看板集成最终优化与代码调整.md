## 一、main.ts 最终完整版本

```typescript
// src/main.ts

import { createApp, type App } from 'vue'
import App from './App.vue'

/**
 * 插件注册顺序（严格遵循依赖关系）：
 *
 * 1. Pinia       — 状态管理基础，必须在最前面
 * 2. Vue Query   — 依赖 Pinia（mutation 回调中可能调用 Store）
 * 3. i18n        — 国际化（依赖 app 实例）
 * 4. Router      — 路由（依赖 Pinia 和 permissionStore）
 * 5. Directives  — 指令（依赖 Pinia Store）
 * 6. Error Handler — 全局错误处理（最后注册，兜底所有异常）
 */

async function bootstrap(): Promise<void> {
  const app: App = createApp(App)

  // ==================== 1. Pinia ====================
  const { setupPinia } = await import('./plugins/pinia')
  setupPinia(app)

  // ==================== 2. TanStack Vue Query ====================
  const { setupVueQuery } = await import('./plugins/vue-query')
  setupVueQuery(app)

  // ==================== 3. Element Plus ====================
  const { setupElementPlus } = await import('./plugins/element-plus')
  setupElementPlus(app)

  // ==================== 4. i18n ====================
  const { setupI18n } = await import('./plugins/i18n')
  setupI18n(app)

  // ==================== 5. Router ====================
  const { setupRouter } = await import('./plugins/router')
  setupRouter(app)

  // ==================== 6. Directives ====================
  const { setupDirectives } = await import('./directives')
  setupDirectives(app)

  // ==================== 7. 全局错误处理 ====================
  setupErrorHandler(app)

  // ==================== 8. 挂载应用 ====================
  app.mount('#app')

  // 开发环境输出启动信息
  if (import.meta.env.DEV) {
    console.log(
      `%c Vue3 Admin Pro %c v1.0.0 `,
      'background:#409eff;color:#fff;padding:4px 8px;border-radius:3px 0 0 3px;',
      'background:#67c23a;color:#fff;padding:4px 8px;border-radius:0 3px 3px 0;'
    )
  }
}

/**
 * 全局错误处理器
 * - 捕获 Vue 组件渲染错误
 * - 捕获 TanStack Query 未处理的错误
 * - 生产环境上报至错误监控平台（如 Sentry）
 */
function setupErrorHandler(app: App): void {
  app.config.errorHandler = (err, instance, info) => {
    console.error(`[Vue Error] ${info}:`, err)

    // 生产环境：上报错误
    if (import.meta.env.PROD) {
      // TODO: 集成 Sentry / LogRocket 等错误监控
      // Sentry.captureException(err)
    }
  }

  // 捕获未处理的 Promise 异常
  window.addEventListener('unhandledrejection', (event) => {
    console.error('[Unhandled Promise]:', event.reason)
    event.preventDefault()
  })

  // 捕获全局 JS 错误
  window.addEventListener('error', (event) => {
    console.error('[Global Error]:', event.error)
  })
}

// 启动应用
bootstrap().catch((err) => {
  console.error('应用启动失败:', err)
  // 展示降级页面
  const root = document.getElementById('app')
  if (root) {
    root.innerHTML = `
      <div style="display:flex;justify-content:center;align-items:center;height:100vh;
                  font-family:system-ui;color:#f56c6c;flex-direction:column;gap:16px;">
        <h2>应用启动失败</h2>
        <p style="color:#909399;">请刷新页面重试，或联系管理员</p>
        <button onclick="location.reload()"
                style="padding:8px 24px;border:1px solid #dcdfe6;border-radius:4px;
                       cursor:pointer;background:#409eff;color:#fff;border-color:#409eff;">
          刷新页面
        </button>
      </div>
    `
  }
})
```

### 1.1 异步启动的好处

| 优势 | 说明 |
|------|------|
| **代码分割** | 各插件通过 `await import()` 动态导入，减少首屏 JS 体积 |
| **错误隔离** | 某个插件加载失败不会阻塞其他插件 |
| **启动降级** | `bootstrap().catch()` 捕获启动异常，展示友好的降级 UI |
| **Tree Shaking** | 未实际使用的插件不会被打包 |

### 2.4 `plugins/router.ts` — 最终版

```typescript
// src/plugins/router.ts

import { createRouter, createWebHistory, type RouteRecordRaw } from 'vue-router'
import type { App } from 'vue'
import { constantRoutes } from '@/router/routes'
import { setupRouterGuards } from '@/router/guards'

/**
 * 创建 Vue Router 实例
 */
export const router = createRouter({
  history: createWebHistory(import.meta.env.VITE_PUBLIC_PATH || '/'),
  routes: constantRoutes as RouteRecordRaw[],
  scrollBehavior() {
    return { top: 0, left: 0 }
  },
})

/**
 * 初始化路由（在 main.ts 中调用）
 */
export function setupRouter(app: App): void {
  setupRouterGuards(router)
  app.use(router)
}

/**
 * 重置路由（登出时调用）
 */
export function resetRouter(): void {
  const routeNames = router.getRoutes().map((r) => r.name)
  constantRoutes.forEach((route) => {
    const idx = routeNames.indexOf(route.name)
    if (idx !== -1) {
      routeNames.splice(idx, 1)
    }
  })
  routeNames.forEach((name) => {
    if (name) router.removeRoute(name)
  })
}
```

### 3.2 `router/guards.ts` — 优化版

```typescript
// src/router/guards.ts

import type { Router } from 'vue-router'
import { getAccessToken } from '@/utils/token'
import { useAuthStore } from '@/stores/auth'
import { usePermissionStore } from '@/stores/permission'
import { useAppStore } from '@/stores/app'
import { asyncRoutes } from './routes'
import NProgress from 'nprogress'
import 'nprogress/nprogress.css'

NProgress.configure({ showSpinner: false })

const whiteList = ['/auth/login', '/auth/register', '/404', '/403']

/**
 * 全局路由导航守卫
 *
 * 执行顺序（严格按以下步骤）：
 * 1. 进度条开始
 * 2. 设置页面标题
 * 3. Token 检查（纯函数，不依赖 Pinia）
 * 4. 已登录 → 加载用户信息 → 获取权限 → 生成动态路由
 * 5. 未登录 → 白名单放行 / 跳转登录页
 */
export function setupRouterGuards(router: Router): void {
  router.beforeEach(async (to, _from, next) => {
    NProgress.start()

    // 页面标题
    document.title = to.meta?.title
      ? `${to.meta.title as string} — 安全看板`
      : '安全看板'

    const hasToken = getAccessToken()

    if (hasToken) {
      // 已登录 → 访问登录页 → 重定向首页
      if (to.path === '/auth/login') {
        next({ path: '/', replace: true })
        NProgress.done()
        return
      }

      const authStore = useAuthStore()
      const permStore = usePermissionStore()
      const appStore = useAppStore()

      // 开启全局 Loading（路由切换时）
      appStore.setGlobalLoading(true)

      // 用户信息未加载
      if (!authStore.userInfo) {
        try {
          await authStore.fetchUserInfo()
        } catch {
          await authStore.logoutAction()
          next(`/auth/login?redirect=${encodeURIComponent(to.fullPath)}`)
          NProgress.done()
          return
        }
      }

      // 动态路由未生成
      if (!permStore.isRoutesGenerated) {
        try {
          await permStore.fetchPermissions()
          const accessedRoutes = permStore.generateRoutes(asyncRoutes)
          accessedRoutes.forEach((route) => router.addRoute(route))
          // 重新导航以匹配新路由
          next({ ...to, replace: true })
          return
        } catch {
          await authStore.logoutAction()
          next(`/auth/login?redirect=${encodeURIComponent(to.fullPath)}`)
          NProgress.done()
          return
        }
      }

      // 更新面包屑
      updateBreadcrumbs(to)
      // 添加标签页
      addVisitedView(to)

      next()
    } else {
      // 未登录
      if (whiteList.includes(to.path)) {
        next()
      } else {
        next(`/auth/login?redirect=${encodeURIComponent(to.fullPath)}`)
        NProgress.done()
      }
    }
  })

  router.afterEach(() => {
    NProgress.done()
    const appStore = useAppStore()
    appStore.setGlobalLoading(false)
  })

  router.onError((error) => {
    console.error('[Router Error]:', error)
    NProgress.done()
  })
}

// ==================== 辅助函数 ====================

function updateBreadcrumbs(to: any): void {
  const appStore = useAppStore()
  const crumbs: Array<{ title: string; path?: string }> = [{ title: '首页', path: '/' }]

  to.matched.forEach((route: any) => {
    if (route.meta?.title && route.path !== '/') {
      crumbs.push({ title: route.meta.title, path: route.path })
    }
  })

  appStore.setBreadcrumbs(crumbs)
}

function addVisitedView(to: any): void {
  const appStore = useAppStore()
  if (to.meta?.hidden) return
  appStore.addVisitedView({
    name: to.name as string,
    path: to.path,
    title: (to.meta.title as string) || '未命名',
  })
}
```

```typescript
// src/stores/auth.ts — logoutAction 中的路由重置集成

async function logoutAction(): Promise<void> {
  try {
    await logout()
  } catch {
    // 忽略后端登出失败
  } finally {
    // 1. 清除 Token
    clearTokens()
    accessToken.value = null
    userInfo.value = null
    permissions.value = []

    // 2. 重置权限（动态路由）
    const { usePermissionStore } = await import('./permission')
    const permStore = usePermissionStore()
    permStore.resetPermissions()

    // 3. 重置路由（移除动态添加的路由）
    const { resetRouter } = await import('@/router')
    resetRouter()

    // 4. 清空标签页和面包屑
    const { useAppStore } = await import('./app')
    const appStore = useAppStore()
    appStore.visitedViews = []
    appStore.breadcrumbs = []

    // 5. 跳转登录页
    const router = useRouter()
    await router.replace('/auth/login')
  }
}
```

### 5.2 组件级优化

```typescript
// 大数据图表组件的性能守卫

import { defineComponent, h } from 'vue'

/**
 * 高开销组件包装器 — 仅在可视区域内渲染
 * 用于 TrendChart 等含 ECharts 的组件
 */
export function withLazyRender(component: any) {
  return defineComponent({
    name: `Lazy_${component.name || 'Component'}`,
    setup(props: any, { slots }: any) {
      const isVisible = ref(false)
      const containerRef = ref<HTMLElement>()

      onMounted(() => {
        const observer = new IntersectionObserver(
          ([entry]) => {
            if (entry.isIntersecting) {
              isVisible.value = true
              observer.disconnect() // 一旦可见就停止监听
            }
          },
          { rootMargin: '200px' } // 提前 200px 开始渲染
        )
        if (containerRef.value) observer.observe(containerRef.value)
      })

      onUnmounted(() => observer.disconnect())

      return () => h('div', { ref: containerRef },
        isVisible.value ? h(component, props, slots) : null
      )
    },
  })
}
```

| TanStack Query 不刷新 | 数据陈旧 | `queryKey` 未包含筛选参数变化 | 确保 `queryKey` 使用 `computed(() => [...])` 包裹响应式参数；检查 `staleTime` 是否过长 |
| Element Plus 样式丢失 | 组件渲染但无样式 | `unplugin-vue-components` 未正确配置 `ElementPlusResolver` | 确认 `resolvers: [ElementPlusResolver({ importStyle: 'sass' })]` 已配置；手动注册的组件也需要引入样式 |
| 动态路由 404 | 刷新页面后 404 | 动态路由在刷新时未重新生成 | 确认路由守卫中 `isRoutesGenerated` 判断正确；刷新时重新调用 `fetchPermissions()` + `generateRoutes()` |
| keep-alive 不生效 | 页面切换后状态丢失 | 组件 `name` 与路由 `name` 不一致 | Vue 组件必须定义 `name` 属性，且与路由配置的 `name` 完全匹配 |
| Token 过期白屏 | 请求返回 401 后页面无反应 | 拦截器中刷新 Token 失败但未处理 | 确认 `api/client.ts` 中 Token 刷新失败后调用 `clearTokens()` + `window.location.href = '/auth/login'` |
| 导出文件乱码 | Excel 内容为乱码 | 响应 `Content-Type` 不正确 | 后端确保返回 `application/vnd.openxmlformats-officedocument.spreadsheetml.sheet`；前端使用 Blob 下载 |
| 图表 resize 不生效 | 窗口缩放后图表变形 | resize 事件监听未正确绑定 | 在 `onUnmounted` 中 `removeEventListener`；使用 ECharts `dispose()` 销毁旧实例 |
| 循环依赖警告 | Madge 报告循环 | Store 之间互相引用 | 将共享逻辑抽到 `composables/` 或 `utils/`；Store 只保留状态，不互相 `import` |
| 自动导入报 `no-undef` | ESLint 报 `ref is not defined` | ESLint 未读取 auto-imports 配置 | 确认 `.eslintrc.cjs` 中 `extends` 包含 `./.eslintrc-auto-import.json`；或禁用相关 ESLint 规则 |
