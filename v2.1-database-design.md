## 一、设计原则

### 1.1 增量迁移原则

| 原则 | 说明 |
|------|------|
| 不删除 | 不删除已有表和字段 |
| 不修改 | 不修改已有表结构 |
| 仅新增 | 仅新增表和字段 |
| 兼容 | 新表与旧表共存，支持逐步切换 |

### 1.2 统一设计规范

| 规范 | 说明 |
|------|------|
| 主键 | BIGSERIAL |
| 审计字段 | created_at / updated_at / is_deleted |
| 软删除 | is_deleted BOOLEAN DEFAULT FALSE |
| 时间戳 | TIMESTAMP DEFAULT NOW() |
| 外键 | REFERENCES table(id) ON DELETE RESTRICT |

## 二、新增数据表

### 2.1 hse_org_unit - 管理组织单元

```sql
CREATE TABLE IF NOT EXISTS hse_org_unit (
    id                  BIGSERIAL PRIMARY KEY,
    parent_id           BIGINT REFERENCES hse_org_unit(id) ON DELETE RESTRICT,
    org_name            VARCHAR(100) NOT NULL,
    org_type            VARCHAR(20) NOT NULL,
    -- company / department / center / town_center
    org_code            VARCHAR(50) UNIQUE,
    leader_employee_id  BIGINT REFERENCES employee(id) ON DELETE SET NULL,
    safety_manager_id   BIGINT REFERENCES employee(id) ON DELETE SET NULL,
    headcount           INTEGER NOT NULL DEFAULT 0,
    sort_order          INTEGER NOT NULL DEFAULT 0,
    status              VARCHAR(20) NOT NULL DEFAULT 'active',
    -- active / inactive / dissolved
    description         TEXT,
    created_at          TIMESTAMP NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMP NOT NULL DEFAULT NOW(),
    is_deleted          BOOLEAN NOT NULL DEFAULT FALSE
);

-- 索引
CREATE INDEX IF NOT EXISTS idx_hse_org_unit_parent      ON hse_org_unit(parent_id) WHERE is_deleted = false;
CREATE INDEX IF NOT EXISTS idx_hse_org_unit_type        ON hse_org_unit(org_type) WHERE is_deleted = false;
CREATE INDEX IF NOT EXISTS idx_hse_org_unit_code        ON hse_org_unit(org_code) WHERE is_deleted = false;
CREATE INDEX IF NOT EXISTS idx_hse_org_unit_status      ON hse_org_unit(status) WHERE is_deleted = false;
CREATE INDEX IF NOT EXISTS idx_hse_org_unit_leader      ON hse_org_unit(leader_employee_id) WHERE leader_employee_id IS NOT NULL AND is_deleted = false;

-- 注释
COMMENT ON TABLE hse_org_unit IS '管理组织单元（公司/部门/作业中心/乡镇作业中心）';
COMMENT ON COLUMN hse_org_unit.org_type IS '组织类型：company-公司 department-职能部门 center-作业中心 town_center-乡镇作业中心';
COMMENT ON COLUMN hse_org_unit.org_code IS '组织编码（业务唯一）';
COMMENT ON COLUMN hse_org_unit.leader_employee_id IS '负责人ID';
COMMENT ON COLUMN hse_org_unit.safety_manager_id IS '安全负责人ID';
```

### 2.2 hse_operation_unit - 生产作业单元

```sql
CREATE TABLE IF NOT EXISTS hse_operation_unit (
    id                  BIGSERIAL PRIMARY KEY,
    org_unit_id         BIGINT NOT NULL REFERENCES hse_org_unit(id) ON DELETE RESTRICT,
    unit_name           VARCHAR(100) NOT NULL,
    unit_type           VARCHAR(30) NOT NULL,
    -- transfer_station / collection_team / cleaning_team / barrel_team
    -- vehicle_team / vacuum_vehicle_team / greening_team / maintenance_team
    unit_code           VARCHAR(50) UNIQUE,
    leader_employee_id  BIGINT REFERENCES employee(id) ON DELETE SET NULL,
    headcount           INTEGER NOT NULL DEFAULT 0,
    status              VARCHAR(20) NOT NULL DEFAULT 'active',
    -- active / inactive / disbanded
    description         TEXT,
    created_at          TIMESTAMP NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMP NOT NULL DEFAULT NOW(),
    is_deleted          BOOLEAN NOT NULL DEFAULT FALSE
);

-- 索引
CREATE INDEX IF NOT EXISTS idx_hse_operation_unit_org    ON hse_operation_unit(org_unit_id) WHERE is_deleted = false;
CREATE INDEX IF NOT EXISTS idx_hse_operation_unit_type  ON hse_operation_unit(unit_type) WHERE is_deleted = false;
CREATE INDEX IF NOT EXISTS idx_hse_operation_unit_code  ON hse_operation_unit(unit_code) WHERE is_deleted = false;
CREATE INDEX IF NOT EXISTS idx_hse_operation_unit_status ON hse_operation_unit(status) WHERE is_deleted = false;

-- 注释
COMMENT ON TABLE hse_operation_unit IS '生产作业单元（转运站/收运车组/保洁组/挂桶组等）';
COMMENT ON COLUMN hse_operation_unit.unit_type IS '作业单元类型';
COMMENT ON COLUMN hse_operation_unit.unit_code IS '作业单元编码（业务唯一）';
```

### 2.3 hse_employee_relation - 人员组织关系

```sql
CREATE TABLE IF NOT EXISTS hse_employee_relation (
    id                  BIGSERIAL PRIMARY KEY,
    employee_id         BIGINT NOT NULL REFERENCES employee(id) ON DELETE RESTRICT,
    org_unit_id         BIGINT NOT NULL REFERENCES hse_org_unit(id) ON DELETE RESTRICT,
    operation_unit_id   BIGINT REFERENCES hse_operation_unit(id) ON DELETE SET NULL,
    position_id         BIGINT NOT NULL REFERENCES position(id) ON DELETE RESTRICT,
    supervisor_id       BIGINT REFERENCES employee(id) ON DELETE SET NULL,
    start_date          DATE NOT NULL,
    end_date            DATE,
    -- 当前记录为NULL；调岗后旧记录设为结束日期
    status              VARCHAR(20) NOT NULL DEFAULT '在职',
    -- 在职 / 待岗 / 离职 / 退休 / 返聘
    is_current          BOOLEAN NOT NULL DEFAULT TRUE,
    change_reason       VARCHAR(200),
    created_at          TIMESTAMP NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMP NOT NULL DEFAULT NOW(),
    is_deleted          BOOLEAN NOT NULL DEFAULT FALSE
);

-- 索引
CREATE INDEX IF NOT EXISTS idx_hse_employee_relation_employee      ON hse_employee_relation(employee_id) WHERE is_deleted = false;
CREATE INDEX IF NOT EXISTS idx_hse_employee_relation_current       ON hse_employee_relation(employee_id, is_current) WHERE is_current = true AND is_deleted = false;
CREATE INDEX IF NOT EXISTS idx_hse_employee_relation_org           ON hse_employee_relation(org_unit_id) WHERE is_deleted = false;
CREATE INDEX IF NOT EXISTS idx_hse_employee_relation_operation     ON hse_employee_relation(operation_unit_id) WHERE operation_unit_id IS NOT NULL AND is_deleted = false;
CREATE INDEX IF NOT EXISTS idx_hse_employee_relation_position      ON hse_employee_relation(position_id) WHERE is_deleted = false;
CREATE INDEX IF NOT EXISTS idx_hse_employee_relation_status        ON hse_employee_relation(status) WHERE is_deleted = false;

-- 唯一约束（一个员工同时只有一条当前记录）
DO $$ BEGIN
    CREATE UNIQUE INDEX IF NOT EXISTS uk_hse_employee_relation_current
        ON hse_employee_relation(employee_id) WHERE is_current = true AND is_deleted = false;
EXCEPTION WHEN OTHERS THEN NULL; END $$;

-- 注释
COMMENT ON TABLE hse_employee_relation IS '人员组织关系（员工→管理组织→作业单元→岗位）';
COMMENT ON COLUMN hse_employee_relation.is_current IS '是否当前记录（一个员工同时只有一条is_current=true）';
COMMENT ON COLUMN hse_employee_relation.change_reason IS '变动原因';
```

## 六、视图设计

### 6.1 v_employee_full_info - 员工完整信息视图

```sql
CREATE OR REPLACE VIEW v_employee_full_info AS
SELECT
    e.id AS employee_id,
    e.employee_code,
    e.name,
    e.gender,
    e.birth_date,
    EXTRACT(YEAR FROM age(e.birth_date))::INT AS age,
    e.person_type,
    e.person_category,
    ou.id AS org_unit_id,
    ou.org_name AS org_unit_name,
    ou.org_type AS org_unit_type,
    ou.org_code AS org_unit_code,
    op.id AS operation_unit_id,
    op.unit_name AS operation_unit_name,
    op.unit_type AS operation_unit_type,
    p.id AS position_id,
    p.name AS position_name,
    p.risk_level AS position_risk_level,
    er.start_date,
    er.status AS org_status,
    er.is_current
FROM employee e
LEFT JOIN hse_employee_relation er ON e.id = er.employee_id AND er.is_current = true AND er.is_deleted = false
LEFT JOIN hse_org_unit ou ON er.org_unit_id = ou.id AND ou.is_deleted = false
LEFT JOIN hse_operation_unit op ON er.operation_unit_id = op.id AND op.is_deleted = false
LEFT JOIN position p ON er.position_id = p.id AND p.is_deleted = false
WHERE e.is_deleted = false;

COMMENT ON VIEW v_employee_full_info IS '员工完整信息视图（含组织、作业单元、岗位）';
```

### 6.2 v_org_unit_tree - 组织树视图

```sql
CREATE OR REPLACE VIEW v_org_unit_tree AS
WITH RECURSIVE org_tree AS (
    SELECT
        id,
        parent_id,
        org_name,
        org_type,
        org_code,
        sort_order,
        status,
        ARRAY[id] AS path_ids,
        ARRAY[org_name] AS path_names,
        1 AS level
    FROM hse_org_unit
    WHERE parent_id IS NULL AND is_deleted = false
    UNION ALL
    SELECT
        ou.id,
        ou.parent_id,
        ou.org_name,
        ou.org_type,
        ou.org_code,
        ou.sort_order,
        ou.status,
        ot.path_ids || ou.id,
        ot.path_names || ou.org_name,
        ot.level + 1
    FROM hse_org_unit ou
    JOIN org_tree ot ON ou.parent_id = ot.id
    WHERE ou.is_deleted = false
)
SELECT * FROM org_tree;

COMMENT ON VIEW v_org_unit_tree IS '组织树视图（递归查询）';
```

### 7.2 从 department 表迁移数据

```sql
-- ============================================================
-- V2.1 组织数据迁移脚本
-- 将 department 表数据迁移到 hse_org_unit 表
-- ============================================================

BEGIN;

-- 迁移公司级别（level = 3，当前系统企业级别）
INSERT INTO hse_org_unit (org_name, org_type, org_code, headcount, sort_order, status)
SELECT d.name, 'company', d.dept_code, d.headcount, d.sort_order, d.status
FROM department d
WHERE d.level = 3 AND d.is_deleted = false
AND NOT EXISTS (SELECT 1 FROM hse_org_unit WHERE org_code = d.dept_code);

-- 迁移部门级别（level = 4，职能部门）
INSERT INTO hse_org_unit (parent_id, org_name, org_type, org_code, headcount, sort_order, status)
SELECT
    ou.id AS parent_id,
    d.name,
    CASE
        WHEN d.name LIKE '%作业中心%' THEN 'center'
        ELSE 'department'
    END AS org_type,
    d.dept_code,
    d.headcount,
    d.sort_order,
    d.status
FROM department d
JOIN hse_org_unit ou ON ou.org_type = 'company'
WHERE d.level = 4 AND d.is_deleted = false
AND NOT EXISTS (SELECT 1 FROM hse_org_unit WHERE org_code = d.dept_code);

-- 迁移班组级别（level = 5，乡镇作业中心）
INSERT INTO hse_org_unit (parent_id, org_name, org_type, org_code, headcount, sort_order, status)
SELECT
    ou.id AS parent_id,
    d.name,
    'town_center',
    d.dept_code,
    d.headcount,
    d.sort_order,
    d.status
FROM department d
JOIN hse_org_unit ou ON ou.org_code = 'CENTER-HW'
WHERE d.level = 5 AND d.township IS NOT NULL AND d.is_deleted = false
AND NOT EXISTS (SELECT 1 FROM hse_org_unit WHERE org_code = d.dept_code);

-- 标记 department 表为 deprecated
UPDATE department SET is_deprecated = true WHERE is_deleted = false;

COMMIT;
```

## 八、软删除设计

### 8.1 统一软删除模式

所有新增表使用统一的软删除模式：

| 字段 | 类型 | 默认值 | 说明 |
|------|------|--------|------|
| is_deleted | BOOLEAN | FALSE | 是否已删除 |
| deleted_at | TIMESTAMP | NULL | 删除时间 |

### 8.2 查询规范

所有查询必须包含软删除过滤：

```sql
WHERE is_deleted = false
```
